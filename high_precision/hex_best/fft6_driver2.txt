#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cmath>
#include <vector>
#include <unistd.h>
#include <setjmp.h>
#include <signal.h>

static sigjmp_buf g_jb;
static void on_alarm(int){ siglongjmp(g_jb, 1); }

// natural-order sub-DFT wrappers around production dif/idit (which are bit-reversed)
static void dft_fwd(hint::transform::fft::FFT<double> &fft, double *buf, size_t M){
    fft.dif<false>(buf, M);
    fft.bitrev_cmplx(buf, M);          // dif output is bit-reversed -> make natural
}
static void dft_inv(hint::transform::fft::FFT<double> &fft, double *buf, size_t M){
    fft.bitrev_cmplx(buf, M);          // natural -> bit-reversed for idit input
    fft.idit<false>(buf, M);
}

// fixed 6-step forward (natural-order full DFT)
static void dif6fix(hint::transform::fft::FFT<double> &fft, double *x, size_t N){
    const size_t k = (size_t)log2((double)N);  // N is power of 2
    const size_t N1 = size_t(1) << (k/2), N2 = N >> (k/2);
    const double PI = 3.14159265358979323846;
    std::vector<double> col(2*N1);
    for (size_t b = 0; b < N2; ++b){
        for (size_t a = 0; a < N1; ++a){ col[2*a]=x[2*(a*N2+b)]; col[2*a+1]=x[2*(a*N2+b)+1]; }
        dft_fwd(fft, col.data(), N1);
        for (size_t a = 0; a < N1; ++a){ x[2*(a*N2+b)]=col[2*a]; x[2*(a*N2+b)+1]=col[2*a+1]; }
    }
    for (size_t a = 0; a < N1; ++a)
        for (size_t b = 0; b < N2; ++b){
            double ang = -2*PI*a*b/N; double wr=std::cos(ang), wi=std::sin(ang);
            double *p = x + 2*(a*N2+b); double re=p[0], im=p[1];
            p[0]=re*wr-im*wi; p[1]=re*wi+im*wr;
        }
    for (size_t a = 0; a < N1; ++a) dft_fwd(fft, x + 2*a*N2, N2);
    // transpose R=N1,C=N2
    std::vector<double> tmp(2*N);
    for (size_t a=0;a<N1;++a) for (size_t b=0;b<N2;++b){
        tmp[2*(b*N1+a)]=x[2*(a*N2+b)]; tmp[2*(b*N1+a)+1]=x[2*(a*N2+b)+1];
    }
    memcpy(x, tmp.data(), 2*N*sizeof(double));
}

// fixed 6-step inverse (natural-order full IDFT)
static void idit6fix(hint::transform::fft::FFT<double> &fft, double *x, size_t N){
    const size_t k = (size_t)log2((double)N);
    const size_t N1 = size_t(1) << (k/2), N2 = N >> (k/2);
    const double PI = 3.14159265358979323846;
    std::vector<double> tmp(2*N);
    for (size_t a=0;a<N1;++a) for (size_t b=0;b<N2;++b){
        tmp[2*(a*N2+b)]=x[2*(b*N1+a)]; tmp[2*(a*N2+b)+1]=x[2*(b*N1+a)+1];
    }
    memcpy(x, tmp.data(), 2*N*sizeof(double));
    for (size_t a = 0; a < N1; ++a) dft_inv(fft, x + 2*a*N2, N2);
    for (size_t a = 0; a < N1; ++a)
        for (size_t b = 0; b < N2; ++b){
            double ang = 2*PI*a*b/N; double wr=std::cos(ang), wi=std::sin(ang);
            double *p = x + 2*(a*N2+b); double re=p[0], im=p[1];
            p[0]=re*wr-im*wi; p[1]=re*wi+im*wr;
        }
    std::vector<double> col(2*N1);
    for (size_t b = 0; b < N2; ++b){
        for (size_t a = 0; a < N1; ++a){ col[2*a]=x[2*(a*N2+b)]; col[2*a+1]=x[2*(a*N2+b)+1]; }
        dft_inv(fft, col.data(), N1);
        for (size_t a = 0; a < N1; ++a){ x[2*(a*N2+b)]=col[2*a]; x[2*(a*N2+b)+1]=col[2*a+1]; }
    }
}

static void measure_perm(hint::transform::fft::FFT<double> &fft, size_t N){
    // spike at m=1: dif(delta_1)[j] should equal W^{-pi(j)} -> infer pi(j)
    std::vector<double> x(2*N, 0), y(2*N);
    x[0]=1; x[1]=0; // delta at index 0 gives constant -> use m=1: set x[2]=1 (real at index1)
    std::vector<double> xd(2*N,0); xd[2]=1; xd[3]=0;
    memcpy(y.data(), xd.data(), 2*N*sizeof(double));
    fft.dif<false>(y.data(), N);
    printf("perm N=%zu: ", N);
    for (size_t j=0;j<N;++j){
        double ph = std::atan2(y[2*j+1], y[2*j]); // phase of W^{-pi(j)*1}
        long pj = std::lround(-ph * N / (2*M_PI));
        pj = ((pj % (long)N) + (long)N) % (long)N;
        printf("%ld ", pj);
    }
    printf("\n");
}

int main(){
    const size_t sizes[] = {256,512,1024,2048,4096,8192,16384,32768,65536,131072,262144,524288,1048576,2097152};
    signal(SIGALRM, on_alarm);
    // measure permutation for small N
    {
        hint::transform::fft::FFT<double> fft;
        fft.expand(8192);
        measure_perm(fft, 64);
        measure_perm(fft, 256);
    }
    for (size_t N : sizes){
        int tsec = (N <= 65536) ? 20 : (N <= 1048576 ? 60 : 120);
        double maxd=0, maxrt=0;
        int r;
        if ((r = sigsetjmp(g_jb, 1)) == 0){
            alarm(tsec);
            hint::transform::fft::FFT<double> fft;
            std::vector<double> o(2*N), a(2*N), b(2*N), c(2*N);
            uint64_t s = 0x9E3779B97F4A7C15ULL ^ (N * 2654435761ULL);
            auto rnd = [&](){ s ^= s<<13; s ^= s>>7; s ^= s<<17;
                return (double)((int64_t)(s & 0x7fffffffffffffffULL)) / (double)0x7fffffffffffffffULL * 2.0 - 1.0; };
            for (size_t i=0;i<2*N;++i) o[i] = rnd();
            memcpy(a.data(), o.data(), 2*N*sizeof(double));
            memcpy(b.data(), o.data(), 2*N*sizeof(double));
            fft.expand(N);
            fft.dif<false>(b.data(), N);          // reference (bit-reversed order)
            dif6fix(fft, a.data(), N);            // fixed natural-order 6-step
            // maxd: compare bit-reversed 6-step vs bit-reversed dif
            std::vector<double> ab = a; fft.bitrev_cmplx(ab.data(), N);
            maxd = 0;
            for (size_t i=0;i<2*N;++i){ double d = std::fabs(ab[i]-b[i]); if (d>maxd) maxd=d; }
            // round trip: natural dif6fix -> natural idit6fix == identity
            memcpy(c.data(), a.data(), 2*N*sizeof(double));
            idit6fix(fft, c.data(), N);
            maxrt = 0;
            for (size_t i=0;i<2*N;++i){ double d = std::fabs(c[i]-o[i]); if (d>maxrt) maxrt=d; }
            alarm(0);
            printf("N=%-8zu max|dif6fix-dif|=%.3e max|rt|=%.3e\n", N, maxd, maxrt);
        } else {
            printf("N=%-8zu HANG\n", N);
        }
        fflush(stdout);
    }
    return 0;
}
